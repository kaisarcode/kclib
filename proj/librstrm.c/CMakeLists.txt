cmake_minimum_required(VERSION 3.14)
project(rstrm C)
set(CMAKE_C_STANDARD 11)
if(NOT DEFINED RSTRM_BUILD_VERSION)
    set(RSTRM_BUILD_VERSION 0)
endif()
option(RSTRM_BUILD_TESTS "Build rstrm tests" OFF)
if(NOT MSVC)
    add_compile_options(-O3 -Wall -Wextra -Werror)
    if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
        add_compile_options(-Wno-unused-command-line-argument)
    endif()
    if(APPLE)
        add_definitions(-D_DARWIN_C_SOURCE -D_GNU_SOURCE)
    else()
        add_definitions(-D_GNU_SOURCE)
    endif()
else()
    add_compile_options(/O2 /W3 /WX)
endif()
add_compile_definitions(KC_RSTRM_BUILD_VERSION=${RSTRM_BUILD_VERSION})
set(RSTRM_SOURCES src/librstrm.c lib/kcp/ikcp.c)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    set_source_files_properties(lib/kcp/ikcp.c PROPERTIES COMPILE_OPTIONS "-w")
endif()
set(RSTRM_TEST_EXTERNAL_ARTIFACTS OFF)
if(RSTRM_BUILD_TESTS AND RSTRM_TEST_SHARED_LIBRARY)
    set(RSTRM_TEST_EXTERNAL_ARTIFACTS ON)
endif()
if(NOT WIN32)
    find_package(Threads REQUIRED)
endif()
if(NOT RSTRM_TEST_EXTERNAL_ARTIFACTS)
    add_library(rstrm_static STATIC ${RSTRM_SOURCES})
    set_target_properties(rstrm_static PROPERTIES OUTPUT_NAME "rstrm")
    target_include_directories(rstrm_static PUBLIC src PRIVATE lib/kcp)
    add_library(rstrm_shared SHARED ${RSTRM_SOURCES})
    set_target_properties(rstrm_shared PROPERTIES OUTPUT_NAME "rstrm" WINDOWS_EXPORT_ALL_SYMBOLS ON)
    target_include_directories(rstrm_shared PUBLIC src PRIVATE lib/kcp)
    if(NOT WIN32)
        target_link_libraries(rstrm_static PUBLIC Threads::Threads)
        target_link_libraries(rstrm_shared PUBLIC Threads::Threads)
    endif()
endif()
if(RSTRM_BUILD_TESTS)
    enable_testing()
    if(RSTRM_TEST_SHARED_LIBRARY)
        add_library(rstrm_test_shared SHARED IMPORTED)
        set_target_properties(rstrm_test_shared PROPERTIES IMPORTED_LOCATION "${RSTRM_TEST_SHARED_LIBRARY}")
        if(RSTRM_TEST_IMPORT_LIBRARY)
            set_target_properties(rstrm_test_shared PROPERTIES IMPORTED_IMPLIB "${RSTRM_TEST_IMPORT_LIBRARY}")
        endif()
        set(RSTRM_TEST_SHARED_TARGET rstrm_test_shared)
    else()
        set(RSTRM_TEST_SHARED_TARGET rstrm_shared)
    endif()
    add_executable(rstrm_contract_test src/test.c)
    target_include_directories(rstrm_contract_test PRIVATE src)
    target_link_libraries(rstrm_contract_test PRIVATE ${RSTRM_TEST_SHARED_TARGET})
    if(NOT WIN32)
        target_link_libraries(rstrm_contract_test PRIVATE Threads::Threads)
    endif()
    add_custom_command(TARGET rstrm_contract_test POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
            $<TARGET_FILE:${RSTRM_TEST_SHARED_TARGET}>
            $<TARGET_FILE_DIR:rstrm_contract_test>)
endif()
# Build the command-line frontend against the shared library.
if(TARGET rstrm_shared AND NOT EMSCRIPTEN AND
    NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
    add_executable(rstrm_cli src/rstrm.c)
    set_target_properties(rstrm_cli PROPERTIES
        OUTPUT_NAME "rstrm"
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_ARCHIVE_OUTPUT_DIRECTORY}"
    )
    target_include_directories(rstrm_cli PRIVATE src)
    target_link_libraries(rstrm_cli PRIVATE rstrm_shared)
    if(APPLE)
        set_target_properties(rstrm_cli PROPERTIES
            BUILD_WITH_INSTALL_RPATH TRUE
            INSTALL_RPATH "@loader_path"
        )
    elseif(UNIX)
        set_target_properties(rstrm_cli PROPERTIES
            BUILD_WITH_INSTALL_RPATH TRUE
            INSTALL_RPATH "$ORIGIN"
        )
    endif()
endif()
